local GamemodePriority = {
	"Raid",
	"Dungeon",
	"Infinity Castle",
}

local PriorityMap = {
	["Raid"] = 1,
	["Dungeon"] = 2,
	["Infinity Castle"] = 3,
}

local function autoCreateGamemodes()
	if gamemodeFold:GetAttribute("IsInGamemode") and LocalPlayer:GetAttribute("InMode") then
		return
	else
		gamemodeFold:SetAttribute("IsInGamemode", false)
	end

	local selected = {}

	for gamemode, key in pairs(scriptSettings.GamemodesTab.SelectedGamemode) do
		table.insert(selected, ga memode)
	end

	if #selected == 0 then return end

	table.sort(selected, function(a, b)
		return PriorityMap[a] < PriorityMap[b]   -- << LINHA 452
	end)

	local chosen = selected[1]

	Framework.Remote:Fire(
		"GamemodeSystem",
		"Create",
		chosen,
		scriptSettings.GamemodesTab[string.gsub("Selected" .. chosen, " ", "")],
		scriptSettings.GamemodesTab[string.gsub("Selected" .. chosen .. "Difficulty", " ", "")],
		LocalPlayer.UserId
	)

	if LocalPlayer:GetAttribute("InMode") then
		gamemodeFold:SetAttribute("GamemodeType", chosen)
		gamemodeFold:SetAttribute("IsInGamemode", true)
	else
		gamemodeFold:SetAttribute("GamemodeType", "")
		gamemodeFold:SetAttribute("IsInGamemode", false)
	end
end
